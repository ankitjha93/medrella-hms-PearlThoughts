    %% ==========================================
    %% 1. USER MODULE
    %% ==========================================
    USERS {
        uuid id PK "Primary Key"
        varchar email "Unique login email"
        varchar phone "Contact phone"
        varchar password_hash "Hashed credentials"
        varchar role "DOCTOR | PATIENT | ADMIN"
        varchar auth_provider "LOCAL | GOOGLE"
        boolean is_verified "Verification flag"
        timestamp created_at "Created timestamp"
        timestamp updated_at "Updated timestamp"
    }
    VERIFICATION_TOKENS {
        uuid id PK "Primary Key"
        uuid user_id FK "Reference to USERS"
        varchar token "OTP / Token hash"
        varchar token_type "EMAIL_VERIFY | OTP_LOGIN"
        timestamp expires_at "Expiration datetime"
        boolean is_used "Consumed flag"
        timestamp created_at "Created timestamp"
    }

    %% ==========================================
    %% 2. PATIENT MODULE
    %% ==========================================
    PATIENTS {
        uuid id PK "Primary Key"
        uuid user_id FK "1:1 link to USERS"
        varchar first_name "Patient first name"
        varchar last_name "Patient last name"
        date date_of_birth "Date of birth"
        varchar gender "MALE | FEMALE | OTHER"
        varchar blood_group "Blood group"
        varchar emergency_contact "Emergency phone"
        timestamp created_at "Created timestamp"
        timestamp updated_at "Updated timestamp"
    }

    %% ==========================================
    %% 3. DOCTOR MODULE
    %% ==========================================
    SPECIALIZATIONS {
        uuid id PK "Primary Key"
        varchar name "Specialty name"
        varchar description "Specialty overview"
    }
    DOCTORS {
        uuid id PK "Primary Key"
        uuid user_id FK "1:1 link to USERS"
        uuid specialization_id FK "Medical specialty"
        varchar first_name "Doctor first name"
        varchar last_name "Doctor last name"
        varchar license_number "License registration"
        int experience_years "Practice years"
        decimal consultation_fee "Fee per booking"
        text bio "Profile summary"
        decimal rating_average "Average ratings"
        boolean is_approved "Admin approval flag"
        timestamp created_at "Created timestamp"
        timestamp updated_at "Updated timestamp"
    }

    %% ==========================================
    %% 4. AVAILABILITY MODULE
    %% ==========================================
    DOCTOR_AVAILABILITY {
        uuid id PK "Primary Key"
        uuid doctor_id FK "Reference to DOCTORS"
        int day_of_week "0=Sun to 6=Sat"
        time start_time "Window start"
        time end_time "Window end"
        int slot_duration_minutes "Duration (min)"
        boolean is_active "Active toggle"
        timestamp created_at "Created timestamp"
    }
    TIME_SLOTS {
        uuid id PK "Primary Key"
        uuid doctor_id FK "Doctor offering slot"
        uuid availability_id FK "Parent schedule"
        date slot_date "Calendar date"
        time start_time "Slot start"
        time end_time "Slot end"
        varchar slot_type "REGULAR | ELASTIC"
        int max_capacity "Maximum bookings"
        int current_bookings "Active bookings"
        varchar status "AVAILABLE | BOOKED"
        timestamp created_at "Created timestamp"
    }

    %% ==========================================
    %% 5. APPOINTMENT MODULE
    %% ==========================================
    APPOINTMENTS {
        uuid id PK "Primary Key"
        uuid patient_id FK "Attending patient"
        uuid doctor_id FK "Attending doctor"
        uuid slot_id FK "Reserved time slot"
        date appointment_date "Scheduled date"
        time start_time "Start time"
        time end_time "End time"
        varchar status "SCHEDULED | CONFIRMED | COMPLETED | CANCELLED | RESCHEDULED"
        text reason "Chief complaint / notes"
        text cancellation_reason "Reason if cancelled"
        timestamp created_at "Created timestamp"
        timestamp updated_at "Updated timestamp"
    }
    RESCHEDULE_HISTORY {
        uuid id PK "Primary Key"
        uuid appointment_id FK "Subject appointment"
        uuid previous_slot_id FK "Replaced slot"
        uuid new_slot_id FK "New reserved slot"
        varchar rescheduled_by "PATIENT | DOCTOR"
        text reschedule_reason "Reason for change"
        timestamp rescheduled_at "Reschedule timestamp"
    }

1. User Module

Authentication, role-based access control (`DOCTOR`, `PATIENT`, `ADMIN`), and Google OAuth token verification.

USERS, VERIFICATION_TOKENS
2. Patient Module

Patient profile information, emergency contacts, date of birth, and medical baseline data.

PATIENTS (1:1 USERS)
3. Doctor Module

Medical credentials, license verification, consultation fees, and medical specialties.

DOCTORS, SPECIALIZATIONS
4. Availability Module

Weekly recurring availability rules and discrete calendar slots supporting elastic scheduling.

DOCTOR_AVAILABILITY, TIME_SLOTS
5. Appointment Module

Patient booking lifecycle, status transitions, cancellation, and full reschedule audit history.

APPOINTMENTS, RESCHEDULE_HISTORY
